/***************************************************************************************************
SECTION 14.6 — PRODUCT × SCORE DECISION MATRIX
****************************************************************************************************

PURPOSE:
    Show where simulated strategy outcomes concentrate across product and credit quality.

BUSINESS INTERPRETATION:
    This is one of the core Module 2 validation outputs.

    Product × score-band analysis helps reviewers identify whether decision outcomes are concentrated
    in intuitive segments and whether product-aware policy behavior interacts coherently with
    borrower-quality segmentation.

WHY PRODUCT × SCORE MATTERS:
    Product type and score band are both major strategy dimensions:
        product type shapes exposure, payment mechanics, LGD, and counteroffer feasibility;
        score band shapes synthetic estimated-risk tiering and borrower profile tendencies.

    Reviewing them jointly prevents portfolio-level averages from hiding important segment-level
    strategy behavior.

METRIC CONVENTIONS:
    approval_rate:
        APPROVE rows / applications

    counteroffer_rate:
        COUNTEROFFER rows / applications

    manual_review_rate:
        MANUAL_REVIEW rows / applications

    decline_rate:
        DECLINE rows / applications

    simulated_offer_exposure_rate:
        APPROVE + COUNTEROFFER rows / applications

    approved_expected_loss_all_apps:
        uses COALESCE(approved_expected_loss_amount, 0) because it describes total simulated
        portfolio impact across all applications.

    approved_expected_loss_offer_rows:
        uses only rows where simulated_offer_exposure_flag = TRUE because it describes the profile
        of simulated positive offer-exposure outcomes.

ESTIMATED_PD TERMINOLOGY:
    estimated_pd remains the finalized Module 1 SQL field name and is interpreted here as the
    synthetic estimated-PD proxy.

***************************************************************************************************/

SELECT
    product_type_policy_key,
    score_band_policy_key,

    COUNT(*) AS applications,

    COUNT(*) FILTER (WHERE decision_outcome = 'APPROVE') AS approval_count,
    COUNT(*) FILTER (WHERE decision_outcome = 'APPROVE')::NUMERIC
        / NULLIF(COUNT(*), 0) AS approval_rate,

    COUNT(*) FILTER (WHERE decision_outcome = 'COUNTEROFFER') AS counteroffer_count,
    COUNT(*) FILTER (WHERE decision_outcome = 'COUNTEROFFER')::NUMERIC
        / NULLIF(COUNT(*), 0) AS counteroffer_rate,

    COUNT(*) FILTER (WHERE decision_outcome = 'MANUAL_REVIEW') AS manual_review_count,
    COUNT(*) FILTER (WHERE decision_outcome = 'MANUAL_REVIEW')::NUMERIC
        / NULLIF(COUNT(*), 0) AS manual_review_rate,

    COUNT(*) FILTER (WHERE decision_outcome = 'DECLINE') AS decline_count,
    COUNT(*) FILTER (WHERE decision_outcome = 'DECLINE')::NUMERIC
        / NULLIF(COUNT(*), 0) AS decline_rate,

    COUNT(*) FILTER (WHERE simulated_offer_exposure_flag IS TRUE) AS simulated_offer_exposure_count,
    COUNT(*) FILTER (WHERE simulated_offer_exposure_flag IS TRUE)::NUMERIC
        / NULLIF(COUNT(*), 0) AS simulated_offer_exposure_rate,

    AVG(credit_score) AS avg_credit_score,
    AVG(estimated_pd) AS avg_synthetic_estimated_pd_proxy,
    AVG(estimated_lgd) AS avg_estimated_lgd,

    AVG(requested_amount) AS avg_requested_amount,
    AVG(COALESCE(approved_amount, 0)) AS avg_approved_amount_all_apps,
    AVG(approved_amount) FILTER (WHERE simulated_offer_exposure_flag IS TRUE)
        AS avg_approved_amount_offer_rows,

    AVG(requested_payment_to_income_ratio) AS avg_requested_payment_to_income_ratio,
    AVG(approved_payment_to_income_ratio) FILTER (WHERE simulated_offer_exposure_flag IS TRUE)
        AS avg_approved_pti_offer_rows,

    AVG(requested_expected_loss_amount) AS avg_requested_expected_loss_amount,
    AVG(COALESCE(approved_expected_loss_amount, 0))
        AS avg_approved_expected_loss_all_apps,
    AVG(approved_expected_loss_amount) FILTER (WHERE simulated_offer_exposure_flag IS TRUE)
        AS avg_approved_expected_loss_offer_rows,

    SUM(requested_expected_loss_amount) AS total_requested_expected_loss_amount,
    SUM(COALESCE(approved_expected_loss_amount, 0)) AS total_approved_expected_loss_amount
FROM credit_decisioning_sim.strategy_decisions
GROUP BY
    product_type_policy_key,
    score_band_policy_key
ORDER BY
    product_type_policy_key,
    CASE score_band_policy_key
        WHEN 'SUPER_PRIME' THEN 1
        WHEN 'PRIME' THEN 2
        WHEN 'NEAR_PRIME' THEN 3
        WHEN 'SUBPRIME' THEN 4
        WHEN 'DEEP_SUBPRIME' THEN 5
        ELSE 99
    END;


/*
Product × score × decision-outcome distribution.

This query expands the matrix by final outcome so reviewers can identify whether a given segment
is dominated by one simulated strategy path.
*/
SELECT
    product_type_policy_key,
    score_band_policy_key,
    decision_outcome,

    COUNT(*) AS row_count,

    ROUND(
        100.0 * COUNT(*)::NUMERIC
        / NULLIF(
            SUM(COUNT(*)) OVER (
                PARTITION BY product_type_policy_key, score_band_policy_key
            ),
            0
        ),
        2
    ) AS pct_within_product_score_segment,

    AVG(estimated_pd) AS avg_synthetic_estimated_pd_proxy,
    AVG(requested_payment_to_income_ratio) AS avg_requested_payment_to_income_ratio,
    AVG(COALESCE(approved_amount, 0)) AS avg_approved_amount_all_apps,
    AVG(COALESCE(approved_expected_loss_amount, 0)) AS avg_approved_expected_loss_all_apps
FROM credit_decisioning_sim.strategy_decisions
GROUP BY
    product_type_policy_key,
    score_band_policy_key,
    decision_outcome
ORDER BY
    product_type_policy_key,
    CASE score_band_policy_key
        WHEN 'SUPER_PRIME' THEN 1
        WHEN 'PRIME' THEN 2
        WHEN 'NEAR_PRIME' THEN 3
        WHEN 'SUBPRIME' THEN 4
        WHEN 'DEEP_SUBPRIME' THEN 5
        ELSE 99
    END,
    CASE decision_outcome
        WHEN 'APPROVE' THEN 1
        WHEN 'COUNTEROFFER' THEN 2
        WHEN 'MANUAL_REVIEW' THEN 3
        WHEN 'DECLINE' THEN 4
        ELSE 99
    END;

/***************************************************************************************************
SECTION 14.7 — SYNTHETIC RISK ORDERING QA
****************************************************************************************************

PURPOSE:
    Validate that synthetic estimated-risk segmentation flows coherently into simulated strategy
    outcomes.

BUSINESS INTERPRETATION:
    Higher synthetic estimated-risk tiers should generally show lower clean approval rates and higher
    manual-review or decline rates.

    Counteroffer behavior may appear in elevated risk tiers where exposure treatment is available and
    final feasibility checks pass.

ESTIMATED_PD BOUNDARY:
    estimated_pd remains the finalized Module 1 SQL field name and is interpreted only as the
    synthetic estimated-PD proxy.

    This QA block reviews directional ordering and strategy sensitivity.
    It does not validate a calibrated default model.

WHY PERFECT MONOTONICITY IS NOT REQUIRED:
    Product mix, requested exposure, affordability, behavioral signals, and counteroffer feasibility
    can create realistic overlap. The validation expectation is broad directional coherence, not
    rigid deterministic sorting.

***************************************************************************************************/

WITH risk_summary AS (
    SELECT
        strategy_risk_tier,

        COUNT(*) AS applications,

        COUNT(*) FILTER (WHERE decision_outcome = 'APPROVE')::NUMERIC
            / NULLIF(COUNT(*), 0) AS approval_rate,

        COUNT(*) FILTER (WHERE decision_outcome = 'COUNTEROFFER')::NUMERIC
            / NULLIF(COUNT(*), 0) AS counteroffer_rate,

        COUNT(*) FILTER (WHERE decision_outcome = 'MANUAL_REVIEW')::NUMERIC
            / NULLIF(COUNT(*), 0) AS manual_review_rate,

        COUNT(*) FILTER (WHERE decision_outcome = 'DECLINE')::NUMERIC
            / NULLIF(COUNT(*), 0) AS decline_rate,

        COUNT(*) FILTER (WHERE simulated_offer_exposure_flag IS TRUE)::NUMERIC
            / NULLIF(COUNT(*), 0) AS simulated_offer_exposure_rate
    FROM credit_decisioning_sim.strategy_decisions
    GROUP BY strategy_risk_tier
),

risk_outcome AS (
    SELECT
        strategy_risk_tier,
        decision_outcome,
        COUNT(*) AS row_count,
        AVG(estimated_pd) AS avg_synthetic_estimated_pd_proxy,
        AVG(requested_payment_to_income_ratio) AS avg_requested_payment_to_income_ratio,
        AVG(requested_expected_loss_amount) AS avg_requested_expected_loss_amount,
        AVG(COALESCE(approved_expected_loss_amount, 0)) AS avg_approved_expected_loss_all_apps
    FROM credit_decisioning_sim.strategy_decisions
    GROUP BY
        strategy_risk_tier,
        decision_outcome
)

SELECT
    o.strategy_risk_tier,
    o.decision_outcome,
    o.row_count,

    ROUND(
        100.0 * o.row_count::NUMERIC / NULLIF(s.applications, 0),
        2
    ) AS pct_within_risk_tier,

    o.avg_synthetic_estimated_pd_proxy,
    o.avg_requested_payment_to_income_ratio,
    o.avg_requested_expected_loss_amount,
    o.avg_approved_expected_loss_all_apps,

    s.approval_rate,
    s.counteroffer_rate,
    s.manual_review_rate,
    s.decline_rate,
    s.simulated_offer_exposure_rate
FROM risk_outcome o
JOIN risk_summary s
    ON s.strategy_risk_tier = o.strategy_risk_tier
ORDER BY
    CASE o.strategy_risk_tier
        WHEN 'LOW_SYNTHETIC_RISK' THEN 1
        WHEN 'MODERATE_SYNTHETIC_RISK' THEN 2
        WHEN 'ELEVATED_SYNTHETIC_RISK' THEN 3
        WHEN 'HIGH_SYNTHETIC_RISK' THEN 4
        WHEN 'SEVERE_SYNTHETIC_RISK' THEN 5
        ELSE 99
    END,
    CASE o.decision_outcome
        WHEN 'APPROVE' THEN 1
        WHEN 'COUNTEROFFER' THEN 2
        WHEN 'MANUAL_REVIEW' THEN 3
        WHEN 'DECLINE' THEN 4
        ELSE 99
    END;


/*
Compact synthetic risk tier summary.

This query gives one row per synthetic risk tier for directional ordering review.
*/
SELECT
    strategy_risk_tier,

    COUNT(*) AS applications,

    AVG(estimated_pd) AS avg_synthetic_estimated_pd_proxy,
    MIN(estimated_pd) AS min_synthetic_estimated_pd_proxy,
    MAX(estimated_pd) AS max_synthetic_estimated_pd_proxy,

    COUNT(*) FILTER (WHERE decision_outcome = 'APPROVE')::NUMERIC
        / NULLIF(COUNT(*), 0) AS approval_rate,

    COUNT(*) FILTER (WHERE decision_outcome = 'COUNTEROFFER')::NUMERIC
        / NULLIF(COUNT(*), 0) AS counteroffer_rate,

    COUNT(*) FILTER (WHERE decision_outcome = 'MANUAL_REVIEW')::NUMERIC
        / NULLIF(COUNT(*), 0) AS manual_review_rate,

    COUNT(*) FILTER (WHERE decision_outcome = 'DECLINE')::NUMERIC
        / NULLIF(COUNT(*), 0) AS decline_rate,

    COUNT(*) FILTER (WHERE simulated_offer_exposure_flag IS TRUE)::NUMERIC
        / NULLIF(COUNT(*), 0) AS simulated_offer_exposure_rate,

    SUM(COALESCE(approved_amount, 0)) AS total_approved_amount,
    SUM(COALESCE(approved_expected_loss_amount, 0)) AS total_approved_expected_loss_amount
FROM credit_decisioning_sim.strategy_decisions
GROUP BY strategy_risk_tier
ORDER BY
    CASE strategy_risk_tier
        WHEN 'LOW_SYNTHETIC_RISK' THEN 1
        WHEN 'MODERATE_SYNTHETIC_RISK' THEN 2
        WHEN 'ELEVATED_SYNTHETIC_RISK' THEN 3
        WHEN 'HIGH_SYNTHETIC_RISK' THEN 4
        WHEN 'SEVERE_SYNTHETIC_RISK' THEN 5
        ELSE 99
    END;


/*
Offered versus declined synthetic estimated-risk comparison.

Expected interpretation:
    offered rows should generally have lower average synthetic estimated-PD proxy than declined rows.

This is a broad directional check, not a calibrated model performance test.
*/
SELECT
    COUNT(*) FILTER (WHERE simulated_offer_exposure_flag IS TRUE) AS simulated_offer_exposure_rows,
    COUNT(*) FILTER (WHERE decision_outcome = 'DECLINE') AS decline_rows,
    COUNT(*) FILTER (WHERE decision_outcome = 'MANUAL_REVIEW') AS manual_review_rows,

    AVG(estimated_pd) FILTER (WHERE simulated_offer_exposure_flag IS TRUE)
        AS avg_synthetic_estimated_pd_proxy_offer_rows,

    AVG(estimated_pd) FILTER (WHERE decision_outcome = 'DECLINE')
        AS avg_synthetic_estimated_pd_proxy_decline_rows,

    AVG(estimated_pd) FILTER (WHERE decision_outcome = 'MANUAL_REVIEW')
        AS avg_synthetic_estimated_pd_proxy_manual_review_rows,

    CASE
        WHEN COUNT(*) FILTER (WHERE simulated_offer_exposure_flag IS TRUE) = 0
          OR COUNT(*) FILTER (WHERE decision_outcome = 'DECLINE') = 0
            THEN 'NOT_APPLICABLE_INSUFFICIENT_OFFER_OR_DECLINE_ROWS'

        WHEN AVG(estimated_pd) FILTER (WHERE simulated_offer_exposure_flag IS TRUE)
           <= AVG(estimated_pd) FILTER (WHERE decision_outcome = 'DECLINE')
            THEN 'SYNTHETIC_RISK_ORDERING_DIRECTIONALLY_ALIGNED'

        ELSE 'SYNTHETIC_RISK_ORDERING_REVIEW_REQUIRED'
    END AS synthetic_risk_ordering_status
FROM credit_decisioning_sim.strategy_decisions;



/***************************************************************************************************
SECTION 14.8 — REQUESTED PTI / AFFORDABILITY SENSITIVITY QA
****************************************************************************************************

PURPOSE:
    Validate that requested-structure affordability pressure affects simulated strategy outcomes.

BUSINESS INTERPRETATION:
    Higher requested PTI buckets should generally show fewer clean approvals and more counteroffer,
    manual review, or decline outcomes.

REQUESTED VS APPROVED PTI:
    requested_payment_to_income_ratio comes from the Module 1 requested structure.

    approved_payment_to_income_ratio is a Module 2 post-exposure-treatment affordability proxy that
    exists only for simulated positive offer-exposure rows.

    Manual-review rows remain unresolved and should not have approved PTI populated.

WHY THIS SECTION USES REQUESTED PTI BUCKETS:
    Strategy routing starts from the original requested structure.
    Counteroffer and approved exposure treatment may later improve affordability for feasible
    COUNTEROFFER rows.

***************************************************************************************************/

WITH pti_summary AS (
    SELECT
        requested_pti_bucket,

        COUNT(*) AS applications,

        COUNT(*) FILTER (WHERE decision_outcome = 'APPROVE')::NUMERIC
            / NULLIF(COUNT(*), 0) AS approval_rate,

        COUNT(*) FILTER (WHERE decision_outcome = 'COUNTEROFFER')::NUMERIC
            / NULLIF(COUNT(*), 0) AS counteroffer_rate,

        COUNT(*) FILTER (WHERE decision_outcome = 'MANUAL_REVIEW')::NUMERIC
            / NULLIF(COUNT(*), 0) AS manual_review_rate,

        COUNT(*) FILTER (WHERE decision_outcome = 'DECLINE')::NUMERIC
            / NULLIF(COUNT(*), 0) AS decline_rate,

        COUNT(*) FILTER (WHERE simulated_offer_exposure_flag IS TRUE)::NUMERIC
            / NULLIF(COUNT(*), 0) AS simulated_offer_exposure_rate
    FROM credit_decisioning_sim.strategy_decisions
    GROUP BY requested_pti_bucket
),

pti_outcome AS (
    SELECT
        requested_pti_bucket,
        decision_outcome,
        COUNT(*) AS row_count,

        AVG(requested_payment_to_income_ratio) AS avg_requested_payment_to_income_ratio,
        AVG(approved_payment_to_income_ratio) FILTER (WHERE simulated_offer_exposure_flag IS TRUE)
            AS avg_approved_pti_offer_rows,

        AVG(estimated_pd) AS avg_synthetic_estimated_pd_proxy,
        AVG(requested_expected_loss_amount) AS avg_requested_expected_loss_amount,
        AVG(COALESCE(approved_expected_loss_amount, 0)) AS avg_approved_expected_loss_all_apps
    FROM credit_decisioning_sim.strategy_decisions
    GROUP BY
        requested_pti_bucket,
        decision_outcome
)

SELECT
    o.requested_pti_bucket,
    o.decision_outcome,
    o.row_count,

    ROUND(
        100.0 * o.row_count::NUMERIC / NULLIF(s.applications, 0),
        2
    ) AS pct_within_requested_pti_bucket,

    o.avg_requested_payment_to_income_ratio,
    o.avg_approved_pti_offer_rows,
    o.avg_synthetic_estimated_pd_proxy,
    o.avg_requested_expected_loss_amount,
    o.avg_approved_expected_loss_all_apps,

    s.approval_rate,
    s.counteroffer_rate,
    s.manual_review_rate,
    s.decline_rate,
    s.simulated_offer_exposure_rate
FROM pti_outcome o
JOIN pti_summary s
    ON s.requested_pti_bucket = o.requested_pti_bucket
ORDER BY
    CASE o.requested_pti_bucket
        WHEN 'LOW_REQUESTED_PTI' THEN 1
        WHEN 'MODERATE_REQUESTED_PTI' THEN 2
        WHEN 'ELEVATED_REQUESTED_PTI' THEN 3
        WHEN 'HIGH_REQUESTED_PTI' THEN 4
        WHEN 'SEVERE_REQUESTED_PTI' THEN 5
        ELSE 99
    END,
    CASE o.decision_outcome
        WHEN 'APPROVE' THEN 1
        WHEN 'COUNTEROFFER' THEN 2
        WHEN 'MANUAL_REVIEW' THEN 3
        WHEN 'DECLINE' THEN 4
        ELSE 99
    END;


/*
Affordability sensitivity status check.

This compares approval rates in lower requested PTI buckets versus higher requested PTI buckets.
It is a directional diagnostic, not a hard acceptance rule.
*/
WITH affordability_rates AS (
    SELECT
        COUNT(*) FILTER (
            WHERE requested_pti_bucket IN ('LOW_REQUESTED_PTI', 'MODERATE_REQUESTED_PTI')
        ) AS low_or_moderate_requested_pti_rows,

        COUNT(*) FILTER (
            WHERE requested_pti_bucket IN ('HIGH_REQUESTED_PTI', 'SEVERE_REQUESTED_PTI')
        ) AS high_or_severe_requested_pti_rows,

        COUNT(*) FILTER (
            WHERE requested_pti_bucket IN ('LOW_REQUESTED_PTI', 'MODERATE_REQUESTED_PTI')
              AND decision_outcome = 'APPROVE'
        )::NUMERIC
        / NULLIF(
            COUNT(*) FILTER (
                WHERE requested_pti_bucket IN ('LOW_REQUESTED_PTI', 'MODERATE_REQUESTED_PTI')
            ),
            0
        ) AS low_or_moderate_requested_pti_approval_rate,

        COUNT(*) FILTER (
            WHERE requested_pti_bucket IN ('HIGH_REQUESTED_PTI', 'SEVERE_REQUESTED_PTI')
              AND decision_outcome = 'APPROVE'
        )::NUMERIC
        / NULLIF(
            COUNT(*) FILTER (
                WHERE requested_pti_bucket IN ('HIGH_REQUESTED_PTI', 'SEVERE_REQUESTED_PTI')
            ),
            0
        ) AS high_or_severe_requested_pti_approval_rate
    FROM credit_decisioning_sim.strategy_decisions
)

SELECT
    low_or_moderate_requested_pti_rows,
    high_or_severe_requested_pti_rows,
    low_or_moderate_requested_pti_approval_rate,
    high_or_severe_requested_pti_approval_rate,

    CASE
        WHEN low_or_moderate_requested_pti_rows = 0
          OR high_or_severe_requested_pti_rows = 0
            THEN 'NOT_APPLICABLE_INSUFFICIENT_PTI_BUCKET_DENSITY'

        WHEN high_or_severe_requested_pti_approval_rate
          <= low_or_moderate_requested_pti_approval_rate
            THEN 'REQUESTED_PTI_AFFORDABILITY_SENSITIVITY_DIRECTIONALLY_ALIGNED'

        ELSE 'REQUESTED_PTI_AFFORDABILITY_SENSITIVITY_REVIEW_REQUIRED'
    END AS requested_pti_affordability_sensitivity_status
FROM affordability_rates;


/*
Counteroffer PTI improvement diagnostic.

Final COUNTEROFFER rows should generally show approved PTI below requested PTI because exposure has
been reduced.
*/
SELECT
    COUNT(*) FILTER (WHERE decision_outcome = 'COUNTEROFFER') AS counteroffer_rows,

    AVG(requested_payment_to_income_ratio) FILTER (WHERE decision_outcome = 'COUNTEROFFER')
        AS avg_requested_pti_counteroffer_rows,

    AVG(approved_payment_to_income_ratio) FILTER (WHERE decision_outcome = 'COUNTEROFFER')
        AS avg_approved_pti_counteroffer_rows,

    AVG(
        requested_payment_to_income_ratio - approved_payment_to_income_ratio
    ) FILTER (WHERE decision_outcome = 'COUNTEROFFER')
        AS avg_pti_reduction_counteroffer_rows,

    COUNT(*) FILTER (
        WHERE decision_outcome = 'COUNTEROFFER'
          AND approved_payment_to_income_ratio >= requested_payment_to_income_ratio
    ) AS counteroffer_rows_without_pti_improvement
FROM credit_decisioning_sim.strategy_decisions;



/***************************************************************************************************
SECTION 14.9 — DTI CAPACITY QA
****************************************************************************************************

PURPOSE:
    Validate that DTI capacity pressure affects simulated strategy routing.

BUSINESS INTERPRETATION:
    Higher DTI should generally reduce clean approval likelihood and increase review or decline
    routing.

DTI BOUNDARY:
    DTI reflects existing-obligation capacity.

    In Module 2 V1.0, DTI is not directly fixed by counteroffer because a counteroffer adjusts the
    requested exposure treatment, not the synthetic applicant's existing obligations.

***************************************************************************************************/

WITH dti_summary AS (
    SELECT
        dti_bucket,

        COUNT(*) AS applications,

        COUNT(*) FILTER (WHERE decision_outcome = 'APPROVE')::NUMERIC
            / NULLIF(COUNT(*), 0) AS approval_rate,

        COUNT(*) FILTER (WHERE decision_outcome = 'COUNTEROFFER')::NUMERIC
            / NULLIF(COUNT(*), 0) AS counteroffer_rate,

        COUNT(*) FILTER (WHERE decision_outcome = 'MANUAL_REVIEW')::NUMERIC
            / NULLIF(COUNT(*), 0) AS manual_review_rate,

        COUNT(*) FILTER (WHERE decision_outcome = 'DECLINE')::NUMERIC
            / NULLIF(COUNT(*), 0) AS decline_rate,

        COUNT(*) FILTER (WHERE simulated_offer_exposure_flag IS TRUE)::NUMERIC
            / NULLIF(COUNT(*), 0) AS simulated_offer_exposure_rate
    FROM credit_decisioning_sim.strategy_decisions
    GROUP BY dti_bucket
),

dti_outcome AS (
    SELECT
        dti_bucket,
        decision_outcome,
        COUNT(*) AS row_count,
        AVG(debt_to_income_ratio) AS avg_debt_to_income_ratio,
        AVG(estimated_pd) AS avg_synthetic_estimated_pd_proxy,
        AVG(requested_payment_to_income_ratio) AS avg_requested_payment_to_income_ratio,
        AVG(requested_expected_loss_amount) AS avg_requested_expected_loss_amount
    FROM credit_decisioning_sim.strategy_decisions
    GROUP BY

SELECT

    o.avg_debt_to_income_ratio,
    o.avg_synthetic_estimated_pd_proxy,
    o.avg_requested_payment_to_income_ratio,
    o.avg_requested_expected_loss_amount,

    s.decline_rate,
    s.simulated_offer_exposure_rate
FROM dti_outcome o
JOIN dti_summary s
    ON s.dti_bucket = o.dti_bucket
ORDER BY
    CASE o.dti_bucket
        WHEN 'LOW_DTI' THEN 1
        WHEN 'MODERATE_DTI' THEN 2
        WHEN 'ELEVATED_DTI' THEN 3
        WHEN 'HIGH_DTI' THEN 4
        WHEN 'SEVERE_DTI' THEN 5
        ELSE 99
    END,
    CASE o.decision_outcome
        WHEN 'APPROVE' THEN 1
        WHEN 'COUNTEROFFER' THEN 2
        WHEN 'MANUAL_REVIEW' THEN 3
        WHEN 'DECLINE' THEN 4
        ELSE 99
    END;


/*
DTI decline-trigger diagnostic.

Rows with severe DTI triggers should be routed consistently with the accepted decision hierarchy.
*/
SELECT
    dti_decline_trigger_flag,
    decision_outcome,
    decision_path,
    primary_reason_code,
    COUNT(*) AS row_count
FROM credit_decisioning_sim.strategy_decisions
GROUP BY
    dti_decline_trigger_flag,
    decision_outcome,
    decision_path,
    primary_reason_code
ORDER BY
    dti_decline_trigger_flag DESC,
    CASE decision_outcome
        WHEN 'DECLINE' THEN 1
        WHEN 'MANUAL_REVIEW' THEN 2
        WHEN 'COUNTEROFFER' THEN 3
        WHEN 'APPROVE' THEN 4
        ELSE 99
    END,
    decision_path,
    primary_reason_code;


/*
DTI sensitivity status check.

This compares approval rates in lower DTI buckets versus higher DTI buckets.
It is a directional diagnostic, not a hard policy assertion.
*/
WITH dti_rates AS (
    SELECT
        COUNT(*) FILTER (
            WHERE dti_bucket IN ('LOW_DTI', 'MODERATE_DTI')
        ) AS low_or_moderate_dti_rows,

        COUNT(*) FILTER (
            WHERE dti_bucket IN ('HIGH_DTI', 'SEVERE_DTI')
        ) AS high_or_severe_dti_rows,

        COUNT(*) FILTER (
            WHERE dti_bucket IN ('LOW_DTI', 'MODERATE_DTI')
              AND decision_outcome = 'APPROVE'
        )::NUMERIC
        / NULLIF(
            COUNT(*) FILTER (
                WHERE dti_bucket IN ('LOW_DTI', 'MODERATE_DTI')
            ),
            0
        ) AS low_or_moderate_dti_approval_rate,

        COUNT(*) FILTER (
            WHERE dti_bucket IN ('HIGH_DTI', 'SEVERE_DTI')
              AND decision_outcome = 'APPROVE'
        )::NUMERIC
        / NULLIF(
            COUNT(*) FILTER (
                WHERE dti_bucket IN ('HIGH_DTI', 'SEVERE_DTI')
            ),
            0
        ) AS high_or_severe_dti_approval_rate
    FROM credit_decisioning_sim.strategy_decisions
)

SELECT
    low_or_moderate_dti_rows,
    high_or_severe_dti_rows,
    low_or_moderate_dti_approval_rate,
    high_or_severe_dti_approval_rate,

    CASE
        WHEN low_or_moderate_dti_rows = 0
          OR high_or_severe_dti_rows = 0
            THEN 'NOT_APPLICABLE_INSUFFICIENT_DTI_BUCKET_DENSITY'

        WHEN high_or_severe_dti_approval_rate <= low_or_moderate_dti_approval_rate
            THEN 'DTI_CAPACITY_SENSITIVITY_DIRECTIONALLY_ALIGNED'

        ELSE 'DTI_CAPACITY_SENSITIVITY_REVIEW_REQUIRED'
    END AS dti_capacity_sensitivity_status
FROM dti_rates;



/***************************************************************************************************
SECTION 14.10 — APPROVED EXPOSURE AND EXPECTED LOSS QA
****************************************************************************************************

PURPOSE:
    Validate final amount, approved-exposure, affordability, and Expected Loss conventions.

BUSINESS INTERPRETATION:
    This section confirms that final approved exposure economics behave as designed:

        APPROVE:
            approved_amount = requested_amount
            approved_expected_loss_amount = requested_expected_loss_amount

        COUNTEROFFER:
            approved_amount < requested_amount
            approved_expected_loss_amount should be reduced or at least not exceed requested-exposure
            Expected Loss under the accepted feasibility logic.

        MANUAL_REVIEW:
            approved exposure fields remain NULL because manual review is unresolved by design.

        DECLINE:
            approved exposure and approved Expected Loss are zero by convention.

REQUESTED VS APPROVED EXPECTED LOSS:
    requested_expected_loss_amount is requested-exposure Expected Loss from Module 1.

    approved_expected_loss_amount is Module 2 approved-exposure Expected Loss after simulated
    exposure treatment.

SIMULATED OFFER EXPOSURE BOUNDARY:
    Approved exposure metrics are simulated strategy outputs.
    They do not imply operational account creation, funding, or real applicant treatment.

***************************************************************************************************/

SELECT
    decision_outcome,

    COUNT(*) AS row_count,

    AVG(requested_amount) AS avg_requested_amount,
    AVG(COALESCE(approved_amount, 0)) AS avg_approved_amount_all_apps,
    AVG(approved_amount) FILTER (WHERE simulated_offer_exposure_flag IS TRUE)
        AS avg_approved_amount_offer_rows,

    SUM(requested_amount) AS total_requested_amount,
    SUM(COALESCE(approved_amount, 0)) AS total_approved_amount,

    AVG(exposure_reduction_pct) FILTER (WHERE decision_outcome = 'COUNTEROFFER')
        AS avg_exposure_reduction_pct_for_counteroffers,

    MIN(exposure_reduction_pct) FILTER (WHERE decision_outcome = 'COUNTEROFFER')
        AS min_exposure_reduction_pct_for_counteroffers,

    MAX(exposure_reduction_pct) FILTER (WHERE decision_outcome = 'COUNTEROFFER')
        AS max_exposure_reduction_pct_for_counteroffers,

    AVG(requested_payment_to_income_ratio) AS avg_requested_payment_to_income_ratio,
    AVG(approved_payment_to_income_ratio) FILTER (WHERE simulated_offer_exposure_flag IS TRUE)
        AS avg_approved_pti_offer_rows,

    SUM(requested_expected_loss_amount) AS total_requested_expected_loss_amount,
    SUM(COALESCE(approved_expected_loss_amount, 0)) AS total_approved_expected_loss_amount,

    AVG(requested_expected_loss_amount) AS avg_requested_expected_loss_amount,
    AVG(COALESCE(approved_expected_loss_amount, 0)) AS avg_approved_expected_loss_all_apps,
    AVG(approved_expected_loss_amount) FILTER (WHERE simulated_offer_exposure_flag IS TRUE)
        AS avg_approved_expected_loss_offer_rows,

    AVG(approved_el_reduction_pct) FILTER (WHERE decision_outcome = 'COUNTEROFFER')
        AS avg_approved_el_reduction_pct_for_counteroffers
FROM credit_decisioning_sim.strategy_decisions
GROUP BY decision_outcome
ORDER BY
    CASE decision_outcome
        WHEN 'APPROVE' THEN 1
        WHEN 'COUNTEROFFER' THEN 2
        WHEN 'MANUAL_REVIEW' THEN 3
        WHEN 'DECLINE' THEN 4
        ELSE 99
    END;


/*
Final approved exposure convention validation.

Expected result:
    all violation counts should be zero.
*/
SELECT
    COUNT(*) FILTER (
        WHERE decision_outcome = 'APPROVE'
          AND approved_amount IS DISTINCT FROM requested_amount
    ) AS approve_approved_amount_violation_rows,

    COUNT(*) FILTER (
        WHERE decision_outcome = 'COUNTEROFFER'
          AND (
                approved_amount IS NULL
             OR approved_amount <= 0
             OR approved_amount >= requested_amount
             OR counteroffer_amount IS DISTINCT FROM approved_amount
             OR exposure_reduction_flag IS DISTINCT FROM TRUE
             OR exposure_reduction_amount IS NULL
             OR exposure_reduction_amount <= 0
             OR exposure_reduction_pct IS NULL
             OR exposure_reduction_pct <= 0
          )
    ) AS counteroffer_approved_amount_violation_rows,

    COUNT(*) FILTER (
        WHERE decision_outcome = 'MANUAL_REVIEW'
          AND (
                approved_amount IS NOT NULL
             OR counteroffer_amount IS NOT NULL
             OR exposure_reduction_amount IS NOT NULL
             OR exposure_reduction_pct IS NOT NULL
          )
    ) AS manual_review_approved_amount_violation_rows,

    COUNT(*) FILTER (
        WHERE decision_outcome = 'DECLINE'
          AND (
                approved_amount IS DISTINCT FROM 0::NUMERIC
             OR counteroffer_amount IS NOT NULL
             OR exposure_reduction_amount IS DISTINCT FROM requested_amount
             OR exposure_reduction_pct IS DISTINCT FROM 1.0::NUMERIC
          )
    ) AS decline_approved_amount_violation_rows,

    COUNT(*) FILTER (
        WHERE approved_amount IS NOT NULL
          AND approved_amount < 0
    ) AS negative_approved_amount_rows,

    COUNT(*) FILTER (
        WHERE approved_amount IS NOT NULL
          AND approved_amount > requested_amount
    ) AS approved_amount_above_requested_amount_rows
FROM credit_decisioning_sim.strategy_decisions;


/*
Final approved Expected Loss convention validation.

Expected result:
    all violation counts should be zero.
*/
SELECT
    COUNT(*) FILTER (
        WHERE decision_outcome = 'APPROVE'
          AND approved_expected_loss_amount IS DISTINCT FROM requested_expected_loss_amount
    ) AS approve_approved_expected_loss_violation_rows,

    COUNT(*) FILTER (
        WHERE decision_outcome = 'COUNTEROFFER'
          AND (
                approved_expected_loss_amount IS NULL
             OR approved_expected_loss_amount < 0
             OR approved_expected_loss_amount > requested_expected_loss_amount
          )
    ) AS counteroffer_approved_expected_loss_violation_rows,

    COUNT(*) FILTER (
        WHERE decision_outcome = 'MANUAL_REVIEW'
          AND (
                approved_expected_loss_amount IS NOT NULL
             OR approved_el_reduction_amount IS NOT NULL
             OR approved_el_reduction_pct IS NOT NULL
          )
    ) AS manual_review_approved_expected_loss_violation_rows,

    COUNT(*) FILTER (
        WHERE decision_outcome = 'DECLINE'
          AND (
                approved_expected_loss_amount IS DISTINCT FROM 0::NUMERIC
             OR approved_el_reduction_amount IS DISTINCT FROM requested_expected_loss_amount
             OR approved_el_reduction_pct IS DISTINCT FROM
                CASE
                    WHEN requested_expected_loss_amount > 0 THEN 1.0::NUMERIC
                    ELSE 0::NUMERIC
                END
          )
    ) AS decline_approved_expected_loss_violation_rows,

    COUNT(*) FILTER (
        WHERE approved_expected_loss_amount IS NOT NULL
          AND approved_expected_loss_amount < 0
    ) AS negative_approved_expected_loss_rows
FROM credit_decisioning_sim.strategy_decisions;

Product-level requested versus approved exposure and Expected Loss profile.

This helps identify which products contribute most to simulated approved exposure and approved
Expected Loss after Module 2 strategy treatment.
*/

    AVG(requested_expected_loss_amount) AS avg_requested_expected_loss_amount,

SECTION 14.11 — COUNTEROFFER VALIDATION DIAGNOSTICS
****************************************************************************************************

PURPOSE:

BUSINESS INTERPRETATION:

        approved PTI at or below the governed counteroffer target;
        approved Expected Loss at or below the product approved-exposure cap;
        a counteroffer-family simulated reason code.

COUNTEROFFER BOUNDARY:
    Counteroffer output is a simulated strategy outcome on synthetic applicants.
    It is not a production offer, operational approval, or real applicant treatment.

WHY PRELIMINARY AND FINAL COUNTEROFFERS ARE BOTH REVIEWED:
    Preliminary COUNTEROFFER routing means a row was routed for reduced-exposure feasibility testing.

    Final COUNTEROFFER means Section 8 confirmed feasibility and Section 9 resolved the row to a

***************************************************************************************************/

    COUNT(*) FILTER (WHERE decision_outcome = 'COUNTEROFFER') AS final_counteroffer_rows,

        AS avg_approved_pti_for_final_counteroffers,

    AVG(
        requested_payment_to_income_ratio - approved_payment_to_income_ratio
    ) FILTER (WHERE decision_outcome = 'COUNTEROFFER')
        AS avg_pti_reduction_for_final_counteroffers,

    AVG(requested_expected_loss_amount) FILTER (WHERE decision_outcome = 'COUNTEROFFER')
        AS avg_requested_expected_loss_for_final_counteroffers,

    AVG(approved_expected_loss_amount) FILTER (WHERE decision_outcome = 'COUNTEROFFER')

/***************************************************************************************************
SECTION 14.17 — FINAL OUTPUT CONVENTION VALIDATION
****************************************************************************************************

PURPOSE:
    Confirm final output conventions hold in the latest-run table.

BUSINESS INTERPRETATION:
    This query produces diagnostic failure counts suitable for validation evidence.

EXPECTED RESULT:
    All violation counts should be zero.

WHY THIS SECTION IS READ-ONLY:
    The latest-run table was materialized in Section 11.
    Section 14 validates the output but does not repair, overwrite, or recalculate it.

***************************************************************************************************/

SELECT
    COUNT(*) FILTER (
        WHERE (
            COALESCE(approval_flag::INT, 0)
            + COALESCE(counteroffer_flag::INT, 0)
            + COALESCE(manual_review_flag::INT, 0)
            + COALESCE(decline_flag::INT, 0)
        ) <> 1
    ) AS final_outcome_flag_count_violation_rows,

    COUNT(*) FILTER (
        WHERE simulated_offer_exposure_flag
              IS DISTINCT FROM (decision_outcome IN ('APPROVE', 'COUNTEROFFER'))
    ) AS simulated_offer_exposure_flag_violation_rows,

    COUNT(*) FILTER (
        WHERE decision_outcome = 'APPROVE'
          AND approved_amount IS DISTINCT FROM requested_amount
    ) AS approve_amount_violation_rows,

    COUNT(*) FILTER (
        WHERE decision_outcome = 'COUNTEROFFER'
          AND (
                approved_amount IS NULL
             OR approved_amount <= 0
             OR approved_amount >= requested_amount
             OR counteroffer_amount IS DISTINCT FROM approved_amount
          )
    ) AS counteroffer_amount_violation_rows,

    COUNT(*) FILTER (
        WHERE decision_outcome = 'MANUAL_REVIEW'
          AND approved_amount IS NOT NULL
    ) AS manual_review_amount_violation_rows,

    COUNT(*) FILTER (
        WHERE decision_outcome = 'DECLINE'
          AND approved_amount IS DISTINCT FROM 0::NUMERIC
    ) AS decline_amount_violation_rows,

    COUNT(*) FILTER (
        WHERE decision_outcome = 'APPROVE'
          AND approved_monthly_payment_proxy IS DISTINCT FROM requested_monthly_payment_proxy
    ) AS approve_payment_violation_rows,

    COUNT(*) FILTER (
        WHERE decision_outcome = 'COUNTEROFFER'
          AND (
                approved_monthly_payment_proxy IS NULL
             OR approved_payment_to_income_ratio IS NULL
             OR approved_payment_to_income_ratio > counteroffer_target_pti
          )
    ) AS counteroffer_affordability_violation_rows,

    COUNT(*) FILTER (
        WHERE decision_outcome = 'MANUAL_REVIEW'
          AND (
                approved_monthly_payment_proxy IS NOT NULL
             OR approved_payment_to_income_ratio IS NOT NULL
          )
    ) AS manual_review_affordability_violation_rows,

    COUNT(*) FILTER (
        WHERE decision_outcome = 'DECLINE'
          AND (
                approved_monthly_payment_proxy IS DISTINCT FROM 0::NUMERIC
             OR approved_payment_to_income_ratio IS DISTINCT FROM 0::NUMERIC
          )
    ) AS decline_affordability_violation_rows,

    COUNT(*) FILTER (
        WHERE decision_outcome = 'APPROVE'
          AND approved_expected_loss_amount IS DISTINCT FROM requested_expected_loss_amount
    ) AS approve_expected_loss_violation_rows,

    COUNT(*) FILTER (
        WHERE decision_outcome = 'COUNTEROFFER'
          AND (
                approved_expected_loss_amount IS NULL
             OR approved_expected_loss_amount < 0
             OR approved_expected_loss_amount > requested_expected_loss_amount
             OR approved_expected_loss_amount > product_approved_el_cap
          )
    ) AS counteroffer_expected_loss_violation_rows,

    COUNT(*) FILTER (
        WHERE decision_outcome = 'MANUAL_REVIEW'
          AND approved_expected_loss_amount IS NOT NULL
    ) AS manual_review_expected_loss_violation_rows,

    COUNT(*) FILTER (
        WHERE decision_outcome = 'DECLINE'
          AND approved_expected_loss_amount IS DISTINCT FROM 0::NUMERIC
    ) AS decline_expected_loss_violation_rows,

    COUNT(*) FILTER (
        WHERE primary_reason_code IS NULL
           OR primary_reason_family IS NULL
           OR triggered_reason_code_list IS NULL
           OR reason_code_count IS NULL
           OR reason_code_count < 1
    ) AS reason_code_completeness_violation_rows,

    COUNT(*) FILTER (
        WHERE (
                decision_outcome = 'APPROVE'
            AND primary_reason_family <> 'APPROVAL'
        )
        OR (
                decision_outcome = 'COUNTEROFFER'
            AND primary_reason_family <> 'COUNTEROFFER'
        )
        OR (
                decision_outcome = 'MANUAL_REVIEW'
            AND primary_reason_family <> 'MANUAL_REVIEW'
        )
        OR (
                decision_outcome = 'DECLINE'
            AND primary_reason_family <> 'POLICY_STOP'
        )
    ) AS reason_family_alignment_violation_rows
FROM credit_decisioning_sim.strategy_decisions;



/***************************************************************************************************
SECTION 14.18 — ARCHIVE AND COMPARISON READINESS RECAP
****************************************************************************************************

PURPOSE:
    Summarize whether the latest run is archived and whether Section 13 matched comparison is
    available.

BUSINESS INTERPRETATION:
    This recap helps users decide whether the current latest-run output is ready for archived strategy
    comparison.

    It does not replace Section 13. It gives Section 14 users a latest-run readiness snapshot.

ARCHIVE BOUNDARY:
    This query does not insert archive rows.
    This query does not delete archive rows.
    This query does not update archive rows.

TIMESTAMP BOUNDARY:
    latest_output_created_ts and archive_inserted_ts are audit metadata.
    They are not deterministic strategy-comparison keys.

***************************************************************************************************/

WITH latest_context AS (
    SELECT
        archive_run_flag,
        strategy_archive_reset_mode,

        strategy_set_name,
        strategy_name,
        strategy_version,

        module1_scenario_set_name,
        module1_scenario_name,
        population_id,

        COUNT(*) AS latest_run_row_count,
        COUNT(DISTINCT application_id) AS latest_run_distinct_application_id_count
    FROM credit_decisioning_sim.strategy_decisions
    GROUP BY
        archive_run_flag,
        strategy_archive_reset_mode,
        strategy_set_name,
        strategy_name,
        strategy_version,
        module1_scenario_set_name,
        module1_scenario_name,
        population_id
),

current_archived_slice AS (
    SELECT
        lc.strategy_set_name,
        lc.strategy_name,
        lc.strategy_version,
        lc.module1_scenario_set_name,
        lc.module1_scenario_name,
        lc.population_id,

        COUNT(ar.application_id) AS current_archived_slice_row_count,
        COUNT(DISTINCT ar.application_id) AS current_archived_slice_distinct_application_id_count,
        MIN(ar.archive_inserted_ts) AS min_archive_inserted_ts,
        MAX(ar.archive_inserted_ts) AS max_archive_inserted_ts
    FROM latest_context lc
    LEFT JOIN credit_decisioning_sim.strategy_decisions_archive ar
        ON ar.strategy_set_name = lc.strategy_set_name
       AND ar.strategy_name = lc.strategy_name
       AND ar.strategy_version = lc.strategy_version
       AND ar.module1_scenario_set_name = lc.module1_scenario_set_name
       AND ar.module1_scenario_name = lc.module1_scenario_name
       AND ar.population_id = lc.population_id
    GROUP BY
        lc.strategy_set_name,
        lc.strategy_name,
        lc.strategy_version,
        lc.module1_scenario_set_name,
        lc.module1_scenario_name,
        lc.population_id
),

comparison_inventory AS (
    SELECT
        lc.strategy_set_name,
        lc.module1_scenario_set_name,
        lc.module1_scenario_name,
        lc.population_id,

        COUNT(DISTINCT (
            ar.strategy_set_name,
            ar.strategy_name,
            ar.strategy_version
        )) AS archived_strategy_slice_count_for_same_set_scenario_population
    FROM latest_context lc
    LEFT JOIN credit_decisioning_sim.strategy_decisions_archive ar
        ON ar.strategy_set_name = lc.strategy_set_name
       AND ar.module1_scenario_set_name = lc.module1_scenario_set_name
       AND ar.module1_scenario_name = lc.module1_scenario_name
       AND ar.population_id = lc.population_id
    GROUP BY
        lc.strategy_set_name,
        lc.module1_scenario_set_name,
        lc.module1_scenario_name,
        lc.population_id
)

SELECT
    lc.archive_run_flag,
    lc.strategy_archive_reset_mode,

    lc.strategy_set_name,
    lc.strategy_name,
    lc.strategy_version,

    lc.module1_scenario_set_name,
    lc.module1_scenario_name,
    lc.population_id,

    lc.latest_run_row_count,
    lc.latest_run_distinct_application_id_count,

    cas.current_archived_slice_row_count,
    cas.current_archived_slice_distinct_application_id_count,
    cas.min_archive_inserted_ts,
    cas.max_archive_inserted_ts,

    ci.archived_strategy_slice_count_for_same_set_scenario_population,

    (
        ci.archived_strategy_slice_count_for_same_set_scenario_population >= 2
    ) AS at_least_two_strategy_slices_available_for_matched_comparison,

    CASE
        WHEN lc.archive_run_flag IS FALSE
            THEN 'LATEST_RUN_NOT_MARKED_FOR_ARCHIVE'

        WHEN cas.current_archived_slice_row_count = lc.latest_run_row_count
         AND cas.current_archived_slice_distinct_application_id_count = lc.latest_run_distinct_application_id_count
         AND ci.archived_strategy_slice_count_for_same_set_scenario_population >= 2
            THEN 'ARCHIVED_AND_COMPARISON_CANDIDATES_AVAILABLE'

        WHEN cas.current_archived_slice_row_count = lc.latest_run_row_count
         AND cas.current_archived_slice_distinct_application_id_count = lc.latest_run_distinct_application_id_count
            THEN 'ARCHIVED_CURRENT_SLICE_ONLY'

        ELSE 'ARCHIVE_READINESS_REVIEW_REQUIRED'
    END AS archive_and_comparison_readiness_status
FROM latest_context lc
LEFT JOIN current_archived_slice cas
    ON cas.strategy_set_name = lc.strategy_set_name
   AND cas.strategy_name = lc.strategy_name
   AND cas.strategy_version = lc.strategy_version
   AND cas.module1_scenario_set_name = lc.module1_scenario_set_name
   AND cas.module1_scenario_name = lc.module1_scenario_name
   AND cas.population_id = lc.population_id
LEFT JOIN comparison_inventory ci
    ON ci.strategy_set_name = lc.strategy_set_name
   AND ci.module1_scenario_set_name = lc.module1_scenario_set_name
   AND ci.module1_scenario_name = lc.module1_scenario_name
   AND ci.population_id = lc.population_id;
